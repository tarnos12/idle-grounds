'use strict';
const fs=require('fs'),zlib=require('zlib');
function decode(file){
  const b=fs.readFileSync(file); let p=8, w,h,bd,ct,il, idat=[], plte=null, trns=null;
  while(p<b.length){const len=b.readUInt32BE(p),t=b.toString('ascii',p+4,p+8),d=b.subarray(p+8,p+8+len);p+=12+len;
    if(t==='IHDR'){w=d.readUInt32BE(0);h=d.readUInt32BE(4);bd=d[8];ct=d[9];il=d[12];}
    else if(t==='IDAT')idat.push(d); else if(t==='PLTE')plte=d; else if(t==='tRNS')trns=d; else if(t==='IEND')break;}
  if(bd!==8||il!==0) throw new Error('unsupported bd/il '+bd+'/'+il);
  const ch={6:4,2:3,0:1,4:2,3:1}[ct]; const raw=zlib.inflateSync(Buffer.concat(idat)); const stride=w*ch;
  const out=Buffer.alloc(w*h*ch); let prev=Buffer.alloc(stride);
  for(let y=0;y<h;y++){const f=raw[y*(stride+1)];const line=raw.subarray(y*(stride+1)+1,(y+1)*(stride+1));const cur=Buffer.alloc(stride);
    for(let x=0;x<stride;x++){const a=x>=ch?cur[x-ch]:0,up=prev[x],c=x>=ch?prev[x-ch]:0;let v=line[x];
      if(f===1)v+=a;else if(f===2)v+=up;else if(f===3)v+=((a+up)>>1);else if(f===4){const pp=a+up-c,pa=Math.abs(pp-a),pb=Math.abs(pp-up),pc=Math.abs(pp-c);v+=(pa<=pb&&pa<=pc)?a:(pb<=pc?up:c);}
      cur[x]=v&255;} cur.copy(out,y*stride); prev=cur;}
  const rgba=Buffer.alloc(w*h*4);
  for(let i=0;i<w*h;i++){ if(ct===6){out.copy(rgba,i*4,i*4,i*4+4);} else if(ct===2){rgba[i*4]=out[i*3];rgba[i*4+1]=out[i*3+1];rgba[i*4+2]=out[i*3+2];rgba[i*4+3]=255;}
    else if(ct===3){const k=out[i];rgba[i*4]=plte[k*3];rgba[i*4+1]=plte[k*3+1];rgba[i*4+2]=plte[k*3+2];rgba[i*4+3]=trns&&k<trns.length?trns[k]:255;}
    else if(ct===0){rgba.fill(out[i],i*4,i*4+3);rgba[i*4+3]=255;} else {rgba.fill(out[i*2],i*4,i*4+3);rgba[i*4+3]=out[i*2+1];}}
  return {w,h,ct,bd,rgba};
}
function crc32(buf){let c,crc=0xffffffff;for(let n=0;n<buf.length;n++){c=(crc^buf[n])&0xff;for(let k=0;k<8;k++)c=c&1?0xedb88320^(c>>>1):c>>>1;crc=(crc>>>8)^c;}return (crc^0xffffffff)>>>0;}
function encode(file,w,h,rgba){const raw=Buffer.alloc(h*(w*4+1));for(let y=0;y<h;y++){raw[y*(w*4+1)]=0;rgba.copy(raw,y*(w*4+1)+1,y*w*4,(y+1)*w*4);}
  const chunk=(t,d)=>{const l=Buffer.alloc(4);l.writeUInt32BE(d.length);const td=Buffer.concat([Buffer.from(t),d]);const c=Buffer.alloc(4);c.writeUInt32BE(crc32(td));return Buffer.concat([l,td,c]);};
  const ih=Buffer.alloc(13);ih.writeUInt32BE(w,0);ih.writeUInt32BE(h,4);ih[8]=8;ih[9]=6;
  fs.writeFileSync(file,Buffer.concat([Buffer.from([137,80,78,71,13,10,26,10]),chunk('IHDR',ih),chunk('IDAT',zlib.deflateSync(raw)),chunk('IEND',Buffer.alloc(0))]));}
module.exports={decode,encode};
